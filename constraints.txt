# Shared Victor development environment constraints.
#
# Use with the shared ../.venv across codingagent and sibling victor-* repos:
#   ../.venv/bin/python -m pip install -c constraints.txt -e ".[dev]"
#
# These caps keep pip from selecting versions that are individually valid but
# incompatible with common optional tools already used in the shared environment.

# datasets 5.x currently requires fsspec[http] <= 2026.6.0.
# Constraints cannot include extras, so constrain fsspec itself.
fsspec>=2023.1.0,<=2026.6.0

# Semgrep 1.x requires peewee ~= 3.14.
peewee>=3.14,<4

# Keep the shared environment on Victor's supported NumPy range. Numba 0.67
# accepts NumPy <2.6, so this remains compatible with its current release.
numpy>=1.24,<2.6

# arxiv 4.x requires requests >=2.32,<2.35.
requests>=2.32,<2.35

# OpenTelemetry packages must move as a family: API/SDK/exporters use 1.x,
# instrumentation packages use matching 0.x beta versions. 1.45.2/0.66b2
# closes the high-severity OpenTelemetry API advisories; install Semgrep in
# its own isolated environment because it still pins its own older family.
opentelemetry-api==1.45.2
opentelemetry-exporter-otlp==1.45.2
opentelemetry-exporter-otlp-proto-common==1.45.2
opentelemetry-exporter-otlp-proto-grpc==1.45.2
opentelemetry-exporter-otlp-proto-http==1.45.2
opentelemetry-proto==1.45.2
opentelemetry-sdk==1.45.2
opentelemetry-instrumentation==0.66b2
opentelemetry-instrumentation-asgi==0.66b2
opentelemetry-instrumentation-fastapi==0.66b2
opentelemetry-instrumentation-requests==0.66b2
opentelemetry-instrumentation-threading==0.66b2
opentelemetry-semantic-conventions==0.66b2
opentelemetry-util-http==0.66b2

# OpenTelemetry instrumentation 0.66b2 requires wrapt <3,>=1.
wrapt>=1.0,<3
